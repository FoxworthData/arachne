class WalmartStoreDirectoryCityParser:
    def __init__(self):
        # Cache for format lookups
        self.format_cache = {}
        
    def extract_stores_from_next_data(self, html_content, db_session):
        """Extract store data with proper StoreFormat mapping"""
        
        stores_data = self.get_stores_from_json(html_content)
        enriched_stores = []
        
        for store_node in stores_data:
            # Get format_id from store format name
            format_name = store_node.get('name')  # "Walmart Supercenter"
            format_id = self.get_format_id(format_name, db_session)
            
            store_entry = {
                'retailer_store_id': store_node.get('id'),
                'format_id': format_id,  # FK to store_formats table
                'display_name': store_node.get('displayName'),
                'address': store_node.get('address', {}).get('addressLineOne'),
                'city': store_node.get('address', {}).get('city'),
                'state': store_node.get('address', {}).get('state'),
                # ... other fields
            }
            enriched_stores.append(store_entry)
            
        return enriched_stores
    
    def get_format_id(self, format_name, db_session):
        """Get format_id with caching for performance"""
        if format_name in self.format_cache:
            return self.format_cache[format_name]
            
        # Query database for format
        store_format = db_session.query(StoreFormat).filter_by(
            name=format_name,
            is_deleted=False
        ).first()
        
        if store_format:
            self.format_cache[format_name] = store_format.format_id
            return store_format.format_id
        else:
            # Handle unknown format - could log warning or create new format
            self.logger.warning(f"Unknown store format: {format_name}")
            return None  # or default format_id

class StoreDirectoryPersistence:
    """Handles store data persistence to database"""
    
    def persist_stores(self, parsed_stores, db_session, logger):
        """Persist parsed store data to database with proper relationships"""
        
        persisted_stores = []
        for store_data in parsed_stores:
            # Get or create store format
            format_id = self.get_or_create_store_format(
                store_data['store_brand_format'], db_session
            )
            
            # Check if store already exists
            existing_store = db_session.query(Store).filter_by(
                retailer_store_id=store_data['store_id']
            ).first()
            
            if existing_store:
                # Update existing store
                self.update_store_data(existing_store, store_data, format_id)
            else:
                # Create new store
                new_store = self.create_new_store(store_data, format_id)
                db_session.add(new_store)
                
            persisted_stores.append(store_data['store_id'])
            
        return persisted_stores
    
    def get_or_create_store_format(self, format_name, db_session):
        """Get format_id, creating if needed"""
        store_format = db_session.query(StoreFormat).filter_by(
            name=format_name,
            brand_id=1,  # Walmart brand_id
            is_deleted=False
        ).first()
        
        if not store_format:
            # Create new format if needed
            store_format = StoreFormat(
                brand_id=1,
                name=format_name
            )
            db_session.add(store_format)
            db_session.flush()  # Get format_id
            
        return store_format.format_id

                        })

# backend/components/walmart/forager/store_directory_complete_fetcher.py
class WalmartStoreDirectoryCompleteFetcher(RetailerFetcher):
    """
    Intelligent two-phase fetcher:
    Phase 1: Fetch state page and extract city list
    Phase 2: Fetch all city pages automatically
    """
    
    def __init__(self, store_identification: dict, fetch_query: str, start_url: str,
                 project_config, logger, **kwargs):
        super().__init__(
            retailer="Walmart",
            store_identification=store_identification,
            fetch_type="store-directory-complete",
            fetch_query=fetch_query,  # State code like "tx"
            start_url=start_url,
            project_config=project_config,
            logger=logger,
            singleton=False,  # Multi-page fetch
            **kwargs
        )
        
        # Inline parser for immediate city extraction
        self.state_parser = WalmartStoreDirectoryStateParser()
    
    async def fetch_all(self, start_url):
        """
        Two-phase fetch process:
        1. Fetch state page
        2. Parse cities immediately  
        3. Fetch all city pages
        """
        
        self.logger.info(f"🏛️ Starting complete store directory fetch for state: {self.fetch_query}")
        
        # Phase 1: Fetch state directory page
        self.logger.info(f"📋 Phase 1: Fetching state directory page...")
        state_result = await self.fetcher.fetch_single_url(
            start_url, 
            self.handle_state_result, 
            page_num=1
        )
        
        if not state_result or not state_result.get('success'):
            self.logger.error("❌ Failed to fetch state directory page")
            return
        
        # Phase 2: Parse state page to get city list
        self.logger.info(f"🔍 Phase 2: Parsing state page to extract cities...")
        cities = await self.extract_cities_from_state_page(state_result['filename'])
        
        if not cities:
            self.logger.error("❌ No cities found in state directory page")
            return
            
        self.logger.info(f"🏙️ Found {len(cities)} cities with Walmart stores")
        
        # Phase 3: Fetch all city directory pages
        self.logger.info(f"🏪 Phase 3: Fetching store data for all {len(cities)} cities...")
        await self.fetch_all_cities(cities)
        
        self.logger.info(f"✅ Complete store directory fetch finished!")
    
    async def extract_cities_from_state_page(self, state_filename):
        """Parse the just-fetched state page to get city URLs"""
        try:
            # Read the HTML file we just saved
            file_path = self.file_storage.get_file_path(state_filename)
            html_content = self.file_storage.get_html_content(file_path)
            
            # Use the state parser to extract cities
            mock_file_record = type('obj', (object,), {
                'filename': state_filename,
                'fetch_query': self.fetch_query
            })
            
            parsed_result = self.state_parser.parse_html(html_content, mock_file_record)
            
            if parsed_result.get('success'):
                return parsed_result['cities']
            else:
                self.logger.error(f"❌ Failed to parse cities: {parsed_result.get('error')}")
                return []
                
        except Exception as e:
            self.logger.error(f"❌ Error extracting cities: {e}", exc_info=True)
            return []
    
    async def fetch_all_cities(self, cities):
        """Fetch all city directory pages"""
        
        # Optional: Add concurrency control
        max_concurrent = self.project_config.get('store_directory', {}).get('max_concurrent_cities', 5)
        semaphore = asyncio.Semaphore(max_concurrent)
        
        async def fetch_single_city(city_data, city_index):
            async with semaphore:
                city_name = city_data['city']
                city_url = city_data['url']
                
                self.logger.info(f"🏪 Fetching city {city_index + 1}/{len(cities)}: {city_name}")
                
                try:
                    result = await self.fetcher.fetch_single_url(
                        city_url,
                        self.handle_city_result,
                        page_num=city_index + 2  # Start from page 2 (state was page 1)
                    )
                    
                    if result and result.get('success'):
                        self.logger.debug(f"✅ Successfully fetched {city_name}")
                    else:
                        self.logger.warning(f"⚠️ Failed to fetch {city_name}")
                        
                except Exception as e:
                    self.logger.error(f"❌ Error fetching {city_name}: {e}")
        
        # Fetch all cities with controlled concurrency
        tasks = [
            fetch_single_city(city_data, i) 
            for i, city_data in enumerate(cities)
        ]
        
        await asyncio.gather(*tasks, return_exceptions=True)
    
    def handle_state_result(self, result: dict, page_num: int, is_retry: bool = False):
        """Handle state directory page result"""
        filename = None
        if not result['error']:
            # Custom filename for state page
            filename = f"Walmart_store-directory-complete_{self.store_identification['retailer_store_id']}_{self.fetch_query}_state_page_{page_num}_{self.get_timestamp()}.html.gz"
            self.file_storage.save_html(filename, result["html"], fetch_type=self.fetch_type)
        
        # Create FetchedFile record
        fetched_file = self.create_fetched_file_record(
            url=result.get('url', self.start_url),
            filename=filename,
            page_num=page_num,
            result=result,
            file_type='state'
        )
        
        self.session.add_fetched_file(fetched_file)
        return {'success': not result['error'], 'filename': filename}
    
    def handle_city_result(self, result: dict, page_num: int, is_retry: bool = False):
        """Handle city directory page result"""
        filename = None
        if not result['error']:
            # Extract city name from URL for filename
            city_name = self.extract_city_from_url(result.get('url', ''))
            filename = f"Walmart_store-directory-complete_{self.store_identification['retailer_store_id']}_{self.fetch_query}_{city_name}_page_{page_num}_{self.get_timestamp()}.html.gz"
            self.file_storage.save_html(filename, result["html"], fetch_type=self.fetch_type)
        
        # Create FetchedFile record
        fetched_file = self.create_fetched_file_record(
            url=result.get('url', ''),
            filename=filename,
            page_num=page_num,
            result=result,
            file_type='city'
        )
        
        self.session.add_fetched_file(fetched_file)
        return {'success': not result['error'], 'filename': filename}

class WalmartStoreDirectoryCompleteParser(ParserStrategy):
    """
    Unified parser that processes all files from a store-directory-complete session:
    - Parses state file to get city metadata
    - Parses all city files to get store data
    - Returns combined results
    """
    
    def get_parser_type(self):
        return "store-directory-complete"
    
    def parse_session_files(self, session_files, session_metadata):
        """
        Parse all files from a complete store directory session
        
        Args:
            session_files: List of ORMFetchedFile objects
            session_metadata: Session information
        """
        
        state_files = [f for f in session_files if '_state_page_' in f.filename]
        city_files = [f for f in session_files if '_state_page_' not in f.filename]
        
        self.logger.info(f"📊 Processing {len(state_files)} state files and {len(city_files)} city files")
        
        # Parse state file for metadata
        state_data = {}
        if state_files:
            state_data = self.parse_state_file(state_files[0])
        
        # Parse all city files for store data
        all_stores = []
        for city_file in city_files:
            city_stores = self.parse_city_file(city_file)
            if city_stores.get('success'):
                all_stores.extend(city_stores['stores'])
        
        return {
            'success': True,
            'state': state_data.get('state', session_metadata.fetch_query),
            'total_cities_found': len(state_data.get('cities', [])),
            'total_cities_fetched': len(city_files),
            'total_stores': len(all_stores),
            'stores': all_stores,
            'state_metadata': state_data,
            'parsed_from_session': session_metadata.fetcher_session_id
        }

                'store_url': store.get('store_url'),
                'has_pharmacy': 'Pharmacy' in str(store.get('store_services', [])),
                'has_fuel': 'Gas' in str(store.get('store_services', [])) or 'Murphy' in str(store.get('store_services', [])),
                'has_grocery_pickup': 'Grocery Pickup' in str(store.get('store_services', [])),
                
                # Source tracking
                'source_city_url': store.get('source_city_url'),
                'source_file': store.get('source_file')
            }
            
            structured_data.append(store_record)
        
        return pd.DataFrame(structured_data)
    
    def validate_store_data(self, df):
        """
        Data quality validation for store directory data
        """
        
        validation_results = {
            'total_records': len(df),
            'missing_store_id': df['retailer_store_id'].isna().sum(),
            'missing_coordinates': df[['store_latitude', 'store_longitude']].isna().any(axis=1).sum(),
            'missing_address': df['store_address'].isna().sum(),
            'invalid_phone': 0,  # Could add phone format validation
            'duplicate_store_ids': df['retailer_store_id'].duplicated().sum()
        }
        
        # Log validation results
        self.logger.info(f"📊 Data Quality Summary: {validation_results}")
        
        # Could add data cleaning here
        # e.g., df = df.dropna(subset=['retailer_store_id'])
        
        return df
    
    def generate_parquet_filename(self, session_metadata):
        """
        Generate consistent Parquet filename for data lake
        Pattern: parsed_Walmart_store-directory-complete_{store_id}_{state}_{session_id}_{fetch_date}.parquet
        """
        
        fetch_date = session_metadata.start_time.strftime('%Y%m%d') if session_metadata.start_time else 'unknown'
        
        return (
            f"parsed_Walmart_store-directory-complete_"
            f"{session_metadata.store_id}_"
            f"{session_metadata.fetch_query}_"
            f"session_{session_metadata.fetcher_session_id}_"
            f"{fetch_date}.parquet"
        )
    
    def save_to_parquet(self, df, filename):
        """
        Save DataFrame to Parquet in data lake with partitioning
        """
        
        # Add partitioning columns for data lake optimization
        df['fetch_year'] = pd.to_datetime(df['fetch_date']).dt.year
        df['fetch_month'] = pd.to_datetime(df['fetch_date']).dt.month
        df['retailer_partition'] = df['retailer']
        
        # Save to configured data lake location
        parsed_save_location = self.project_config.get('parsed_save_location')
        parquet_path = os.path.join(parsed_save_location, 'Walmart', 'store-directory-complete', filename)
        
        # Ensure directory exists
        os.makedirs(os.path.dirname(parquet_path), exist_ok=True)
        
        # Save with optimization for analytics
        df.to_parquet(
            parquet_path,
            engine='pyarrow',
            compression='snappy',
            index=False
        )
        
        self.logger.info(f"💾 Saved store directory data to: {parquet_path}")
        return parquet_path
